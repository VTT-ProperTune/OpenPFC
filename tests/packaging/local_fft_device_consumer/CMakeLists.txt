cmake_minimum_required(VERSION 3.21)
project(local_fft_device_consumer LANGUAGES C CXX)
find_package(OpenPFC CONFIG REQUIRED)
add_executable(local_fft_device_consumer main.cpp)
target_link_libraries(local_fft_device_consumer PRIVATE OpenPFC::openpfc Heffte::Heffte)
if(backend STREQUAL "cuda")
  target_link_libraries(local_fft_device_consumer PRIVATE OpenPFC::openpfc_gpu_kernels CUDA::cudart CUDA::cufft)
elseif(backend STREQUAL "hip")
  target_compile_definitions(local_fft_device_consumer PRIVATE OPENPFC_TEST_HIP)
  target_link_libraries(local_fft_device_consumer PRIVATE OpenPFC::openpfc_hip_kernels hip::host)
else()
  message(FATAL_ERROR "Pass backend=cuda or hip")
endif()
