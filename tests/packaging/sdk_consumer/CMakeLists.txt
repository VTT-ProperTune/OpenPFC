cmake_minimum_required(VERSION 3.21)
project(sdk_consumer LANGUAGES C CXX)
find_package(OpenPFC 0.5 CONFIG REQUIRED)
if(NOT EXISTS "${OpenPFC_BUILD_MANIFEST}")
  message(FATAL_ERROR "Missing installed build manifest")
endif()
file(READ "${OpenPFC_BUILD_MANIFEST}" manifest)
foreach(pair "source_revision;OpenPFC_SOURCE_REVISION" "source_state;OpenPFC_SOURCE_STATE")
  list(GET pair 0 key)
  list(GET pair 1 variable)
  string(JSON value GET "${manifest}" "${key}")
  if(NOT value STREQUAL "${${variable}}")
    message(FATAL_ERROR "Source identity differs between manifest and package config")
  endif()
endforeach()
foreach(language CXX C)
  foreach(field COMPILER_ID COMPILER_VERSION)
    string(JSON value GET "${manifest}" toolchains "${language}" "CMAKE_${language}_${field}")
    if(NOT value STREQUAL "${CMAKE_${language}_${field}}")
      message(FATAL_ERROR "Manifest compiler identity differs: ${language} ${field}")
    endif()
  endforeach()
endforeach()
get_target_property(options OpenPFC::openpfc INTERFACE_COMPILE_OPTIONS)
if(options MATCHES "-O3|--generate-code|--fmad|--expt-relaxed-constexpr")
  message(FATAL_ERROR "Private optimization/device flags leaked into public options: ${options}")
endif()
get_target_property(definitions OpenPFC::openpfc INTERFACE_COMPILE_DEFINITIONS)
foreach(pair "cuda;OpenPFC_ENABLE_CUDA" "hip;OpenPFC_ENABLE_HIP"
             "heffte;OpenPFC_ENABLE_HEFFTE" "hdf5;OPENPFC_HAS_HDF5=1")
  list(GET pair 0 key)
  list(GET pair 1 definition)
  string(JSON enabled GET "${manifest}" capabilities "${key}")
  if(enabled AND NOT definition IN_LIST definitions)
    message(FATAL_ERROR "SDK claims ${key} without its public compile definition")
  elseif(NOT enabled AND definition IN_LIST definitions)
    message(FATAL_ERROR "SDK exports ${definition} but manifest disables it")
  endif()
endforeach()
if(NOT expected)
  set(expected compatible)
endif()
set(arguments)
if(check_manifest)
  list(APPEND arguments MANIFEST "${check_manifest}")
endif()
if(backend)
  list(APPEND arguments BACKEND "${backend}")
endif()
if(require_compatible)
  list(APPEND arguments REQUIRED)
endif()
openpfc_check_compatibility(RESULT result REPORT report ${arguments})
message(STATUS "${report}")
if(NOT result STREQUAL expected)
  message(FATAL_ERROR "Expected ${expected}, got ${result}: ${report}")
endif()
if(reason AND NOT report MATCHES "${reason}")
  message(FATAL_ERROR "Missing diagnostic '${reason}': ${report}")
endif()
add_executable(sdk_consumer main.cpp)
target_link_libraries(sdk_consumer PRIVATE OpenPFC::openpfc)
