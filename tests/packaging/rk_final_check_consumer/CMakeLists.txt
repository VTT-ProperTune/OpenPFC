cmake_minimum_required(VERSION 3.21)
project(rk_final_check_consumer LANGUAGES C CXX)
find_package(OpenPFC CONFIG REQUIRED)
set(backend host CACHE STRING "host, cuda or hip")
if(backend STREQUAL "cuda")
    enable_language(CUDA)
    set_source_files_properties(main.cpp PROPERTIES LANGUAGE CUDA)
elseif(backend STREQUAL "hip")
    enable_language(HIP)
    set_source_files_properties(main.cpp PROPERTIES LANGUAGE HIP)
elseif(NOT backend STREQUAL "host")
    message(FATAL_ERROR "backend must be host, cuda or hip")
endif()
add_executable(rk_final_check_consumer main.cpp)
target_link_libraries(rk_final_check_consumer PRIVATE OpenPFC::openpfc)
set_target_properties(rk_final_check_consumer PROPERTIES CXX_STANDARD 20
    CUDA_STANDARD 20 CUDA_STANDARD_REQUIRED ON HIP_STANDARD 20 HIP_STANDARD_REQUIRED ON)
if(backend STREQUAL "cuda")
    target_compile_definitions(rk_final_check_consumer PRIVATE CHECK_CUDA)
    target_compile_options(rk_final_check_consumer PRIVATE --fmad=false)
    target_link_libraries(rk_final_check_consumer PRIVATE CUDA::cudart)
elseif(backend STREQUAL "hip")
    target_compile_definitions(rk_final_check_consumer PRIVATE CHECK_HIP)
    target_compile_options(rk_final_check_consumer PRIVATE -ffp-contract=off)
    target_link_libraries(rk_final_check_consumer PRIVATE hip::host)
endif()
enable_testing()
add_test(NAME final_check_installed COMMAND rk_final_check_consumer)
add_test(NAME final_check_installed_2ranks COMMAND ${MPIEXEC_EXECUTABLE}
    ${MPIEXEC_NUMPROC_FLAG} 2 $<TARGET_FILE:rk_final_check_consumer>)
set_tests_properties(final_check_installed final_check_installed_2ranks PROPERTIES TIMEOUT 60)
